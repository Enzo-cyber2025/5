cmake_minimum_required(VERSION 3.22)
project(gguf_mobile LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SOURCE} llama-build)
# App consumes selected tokens, not backend logprob/candidate export APIs.
# Partial backend samplers retain their full CPU fallback outputs.
target_compile_definitions(llama PRIVATE GGUF_TOKEN_ONLY_SAMPLING=1)
# v0.4.1 also includes SPIR-V headers from ggml-vulkan.cpp, not only its shader tool.
if(NOT EXISTS "${GGUF_SPIRV_INCLUDE_DIR}/spirv/unified1/spirv.hpp")
    message(FATAL_ERROR "Pinned SPIR-V headers missing")
endif()
target_include_directories(ggml-vulkan SYSTEM PRIVATE ${GGUF_SPIRV_INCLUDE_DIR})
add_subdirectory(${LLAMA_SOURCE}/tools/mtmd mtmd-build EXCLUDE_FROM_ALL)
target_include_directories(mtmd PRIVATE "${LLAMA_SOURCE}/ggml/src")
add_library(aijni SHARED mobile.cpp)
target_compile_features(aijni PRIVATE cxx_std_17)
option(GGUF_EXPERIMENT_MEDIA_PREFIX "Build explicitly opt-in multimodal KV reuse experiment" OFF)
if(GGUF_EXPERIMENT_MEDIA_PREFIX)
    target_compile_definitions(aijni PRIVATE GGUF_EXPERIMENT_MEDIA_PREFIX=1)
endif()
option(GGUF_EXPERIMENT_EXPANDED_WEIGHTS "Build opt-in full-precision Vulkan weight expansion experiment" OFF)
if(GGUF_EXPERIMENT_EXPANDED_WEIGHTS)
    target_compile_definitions(aijni PRIVATE GGUF_EXPERIMENT_EXPANDED_WEIGHTS=1)
    target_include_directories(aijni PRIVATE "${LLAMA_SOURCE}/src")
endif()
option(GGUF_EXPERIMENT_REPACKED_WEIGHTS "Build opt-in lossless Q5 to Q8 GPU layout experiment" OFF)
if(GGUF_EXPERIMENT_REPACKED_WEIGHTS)
    if(GGUF_EXPERIMENT_EXPANDED_WEIGHTS)
        message(FATAL_ERROR "Do not combine distinct weight experiments")
    endif()
    target_compile_definitions(aijni PRIVATE GGUF_EXPERIMENT_REPACKED_WEIGHTS=1)
    target_include_directories(aijni PRIVATE "${LLAMA_SOURCE}/src")
endif()
target_link_libraries(aijni PRIVATE llama llama-common mtmd ggml android log)
target_link_options(aijni PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
set_target_properties(aijni PROPERTIES CXX_VISIBILITY_PRESET hidden)

# Baseline-only probe, independent of ggml-cpu's optional ARM ISA flags.
add_library(ggufcpu SHARED cpu_probe.cpp)
target_compile_features(ggufcpu PRIVATE cxx_std_17)
target_link_options(ggufcpu PRIVATE -Wl,-z,max-page-size=16384)
set_target_properties(ggufcpu PROPERTIES CXX_VISIBILITY_PRESET hidden)
