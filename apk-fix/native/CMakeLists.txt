cmake_minimum_required(VERSION 3.22)
project(gguf_mobile LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SOURCE} llama-build)
add_subdirectory(${LLAMA_SOURCE}/tools/mtmd mtmd-build EXCLUDE_FROM_ALL)
add_library(aijni SHARED mobile.cpp)
target_compile_features(aijni PRIVATE cxx_std_17)
target_link_libraries(aijni PRIVATE llama llama-common mtmd ggml android log)
target_link_options(aijni PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
set_target_properties(aijni PROPERTIES CXX_VISIBILITY_PRESET hidden)
